(function () { try { var key = 'openmeta-theme' var stored = window.localStorage.getItem(key) var isDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches var theme = stored === 'dark' || stored === 'light' ? stored : isDark ? 'dark' : 'light' document.documentElement.dataset.theme = theme document.documentElement.style.colorScheme = theme } catch (e) { // ignore } })()
